{# report_table.html ----------------- Reusable Jinja2 macro for rendering a sortable/filterable report table. Usage in any report template: {% from "report_table.html" import report_table %} {{ report_table(rows, columns) }} Parameters ---------- rows : list of dicts (from logic_reports.py functions) columns : list of dicts, each with keys: key (str) – the dict key in each row label (str) – column header text badge (bool, optional) – wrap value in a badge if truthy overdue_key (str, optional) – if provided, rows where row[overdue_key] > 0 get class row-overdue empty_msg (str, optional) – message shown when rows is empty Example ------- {% from "report_table.html" import report_table %} {{ report_table( rows, [ {"key": "student_name", "label": "Student"}, {"key": "enrolment_no", "label": "Enrolment No"}, {"key": "open_issues", "label": "Open"}, {"key": "overdue_issues", "label": "Overdue", "badge": true}, ], empty_msg="No data found." ) }} #} {% macro report_table(rows, columns, empty_msg="No data found.") %}
{% for col in columns %} {% endfor %} {% if rows %} {% for r in rows %} {# Determine row class: mark overdue rows #} {% set is_overdue = namespace(val=false) %} {% for col in columns %} {% if col.get('overdue_key') and r.get(col.overdue_key, 0) > 0 %} {% set is_overdue.val = true %} {% endif %} {% endfor %} {% for col in columns %} {% endfor %} {% endfor %} {% else %} {% endif %}
{{ col.label }}
{% set val = r[col.key] if col.key in r else '' %} {% if col.get('badge') and val %} {{ val }} {% else %} {{ val if val is not none else '—' }} {% endif %}
{{ empty_msg }}
{% endmacro %}
